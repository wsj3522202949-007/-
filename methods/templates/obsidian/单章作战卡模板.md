<%*
// 单章作战卡 - Templater 模板
// 使用：在 chapters/ 目录右键 → Templater → 选择本模板
// 弹出输入框：章节名
const chapterName = await tp.system.prompt("章节名（如：汇报会上的新甲方）");
if (!chapterName) { throw new Error("章节名不能为空"); }

// 获取当前文件所在目录的上一级（即项目目录）
const currentPath = tp.file.folder(true);
const projectDir = currentPath.replace(/\/chapters$/, "");

// 尝试读取 STATUS.md 获取下一章编号
let chapterNum = 1;
try {
    const statusPath = `${projectDir}/STATUS.md`;
    const statusContent = await tp.file.include(statusPath);
    // 从 STATUS.md 中解析章节号
    const match = statusContent.match(/\|.*?(\d+)\s*\|/);
    if (match) { chapterNum = parseInt(match[1]) + 1; }
} catch(e) {}

const chapterId = `${chapterNum}`.padStart(3, "0");
const fileName = `第${chapterId}章-${chapterName}`;

// 读取项目 README 获取 POV 等元信息
let pov = "待定";
try {
    const readmePath = `${projectDir}/README.md`;
    const readmeContent = await tp.file.include(readmePath);
    const povMatch = readmeContent.match(/pov:\s*(.*)/);
    if (povMatch) { pov = povMatch[1].trim(); }
} catch(e) {}
-%>
